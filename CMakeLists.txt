cmake_minimum_required(VERSION 3.24)
project(SpherePacking VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(SP_SANITIZE "Instrument memory access with AddressSanitizer" OFF)
add_library(sp_geometry cpp/src/geometry.cpp)
target_include_directories(sp_geometry PUBLIC cpp/include)
add_library(sp_packing cpp/src/packing.cpp)
target_link_libraries(sp_packing PUBLIC sp_geometry)
add_library(sp_io cpp/src/io.cpp)
target_link_libraries(sp_io PUBLIC sp_packing)
add_executable(sphere_packing cpp/src/main.cpp)
target_link_libraries(sphere_packing PRIVATE sp_io)
include(CTest)
foreach(unit geometry packing io)
    add_executable(test_${unit} cpp/tests/test_${unit}.cpp)
    target_link_libraries(test_${unit} PRIVATE sp_${unit})
    add_test(NAME ${unit} COMMAND test_${unit})
endforeach()
add_test(NAME cli_help COMMAND sphere_packing --help)
add_test(NAME cli_reject_missing_model COMMAND sphere_packing --radius 0.3)
set_tests_properties(cli_reject_missing_model PROPERTIES WILL_FAIL TRUE)
add_test(NAME cli_reject_bad_number COMMAND sphere_packing --model nonexistent.stl --radius nan)
set_tests_properties(cli_reject_bad_number PROPERTIES WILL_FAIL TRUE)
add_test(NAME cli_reject_unknown_option COMMAND sphere_packing --nonexistent-option)
set_tests_properties(cli_reject_unknown_option PROPERTIES WILL_FAIL TRUE)
foreach(target sp_geometry sp_packing sp_io sphere_packing test_geometry test_packing test_io)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /EHsc /fp:strict)
        if(CMAKE_CXX_COMPILER_ID STREQUAL "IntelLLVM")
            target_compile_options(${target} PRIVATE /Qfma-)
        endif()
        if(SP_SANITIZE)
            target_compile_options(${target} PRIVATE /fsanitize=address /Zi)
            target_link_options(${target} PRIVATE /INCREMENTAL:NO /DEBUG)
        endif()
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
        if(SP_SANITIZE)
            target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
            target_link_options(${target} PRIVATE -fsanitize=address,undefined)
        endif()
    endif()
endforeach()
