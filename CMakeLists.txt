cmake_minimum_required(VERSION 3.24)
project(SpherePacking VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(SP_SANITIZE "Instrument memory access with AddressSanitizer" OFF)
add_library(sp_radii src/radii.cpp)
target_include_directories(sp_radii PUBLIC include)
add_library(sp_geometry src/geometry.cpp)
target_include_directories(sp_geometry PUBLIC include)
add_library(sp_packing src/packing.cpp)
target_link_libraries(sp_packing PUBLIC sp_geometry)
add_library(sp_io src/io.cpp)
target_link_libraries(sp_io PUBLIC sp_packing sp_radii)
add_executable(sphere_packing src/main.cpp)
target_link_libraries(sphere_packing PRIVATE sp_io)
include(CTest)
foreach(unit geometry packing io radii)
    add_executable(test_${unit} tests/test_${unit}.cpp)
    target_link_libraries(test_${unit} PRIVATE sp_${unit})
    add_test(NAME ${unit} COMMAND test_${unit})
endforeach()
add_test(NAME cli_help COMMAND sphere_packing --help)
foreach(case missing_model bad_number unknown_option invalid_scope radius_modes radius_parameters radius_dispersion
    scalar_generator scalar_generator_reverse scalar_before_preset removed_index_range removed_radii_output)
    add_test(NAME cli_reject_${case} COMMAND ${CMAKE_COMMAND}
        "-DEXECUTABLE=$<TARGET_FILE:sphere_packing>" "-DCASE=${case}"
        -P "${CMAKE_CURRENT_SOURCE_DIR}/tests/check_cli_error.cmake")
endforeach()
add_test(NAME cli_generated_radii COMMAND ${CMAKE_COMMAND}
    "-DEXECUTABLE=$<TARGET_FILE:sphere_packing>"
    "-DWORK=${CMAKE_CURRENT_BINARY_DIR}/test_radii_cli"
    -P "${CMAKE_CURRENT_SOURCE_DIR}/tests/check_cli_radii.cmake")
foreach(target sp_geometry sp_packing sp_io sp_radii sphere_packing test_geometry test_packing test_io test_radii)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /EHsc /fp:strict)
        if(CMAKE_CXX_COMPILER_ID STREQUAL "IntelLLVM")
            target_compile_options(${target} PRIVATE /Qfma-)
        endif()
        if(SP_SANITIZE)
            target_compile_options(${target} PRIVATE /fsanitize=address /Zi)
            target_link_options(${target} PRIVATE /INCREMENTAL:NO /DEBUG)
        endif()
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
        if(SP_SANITIZE)
            target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
            target_link_options(${target} PRIVATE -fsanitize=address,undefined)
        endif()
    endif()
endforeach()
